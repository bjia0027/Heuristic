### 问题1: 伪性能感知 - 最关键！
```python
# types.py 第163行 get_performance_score() 的问题实现
def get_performance_score(self) -> float:
    cpu_score = max(0, 1 - self.cpu_usage / 100)      # ❌ 实时CPU使用率
    load_score = max(0, 1 - self.get_load_ratio())    # ❌ 实时负载率
    success_score = self.success_rate                  # ❌ 可能都是1.0
    latency_score = max(0, 1 - min(self.network_latency / 1000, 1))
    
    return (cpu_score * 0.3 + load_score * 0.3 + 
            success_score * 0.3 + latency_score * 0.1)
```

**致命缺陷**:
- ✅ 所有节点初始状态下：`cpu_usage=0`, `current_load=0`, `success_rate=1.0`, `latency≈0`
- ✅ 导致所有节点`get_performance_score() ≈ 1.0`（完全相同！）
- ✅ 算法退化为纯粹的槽位分配：`score = 1.0 * (1 - load_ratio)`

```python
@dataclass
class ServerNode:
    # 现有字段...
    performance_weight: float = 1.0  # ✅ 新增静态性能权重
    
    def get_performance_score(self) -> float:
        """获取性能评分 - 优先使用静态权重"""
        if hasattr(self, 'performance_weight') and self.performance_weight > 0:
            # ✅ 使用配置的静态权重
            return self.performance_weight
        
        # ❌ 原有动态计算（作为fallback）
        cpu_score = max(0, 1 - self.cpu_usage / 100)
        load_score = max(0, 1 - self.get_load_ratio())
        success_score = self.success_rate
        latency_score = max(0, 1 - min(self.network_latency / 1000, 1))
        
        return (cpu_score * 0.3 + load_score * 0.3 + 
                success_score * 0.3 + latency_score * 0.1)
```

**预期效果**: 
- high节点: `score = 1.0 * (1 - load_ratio)`
- medium节点: `score = 0.5 * (1 - load_ratio)` 
- low节点: `score = 0.25 * (1 - load_ratio)`

```python
# dag_heuristic_scheduler_optimized.py 新增方法
def _should_use_dag_scheduling(self, tasks: List[CompileTask]) -> bool:
    """判断是否应该使用DAG调度"""
    
    # 检查任务数量
    if len(tasks) < 10:
        return False  # 小批量任务用简单调度
    
    # 检查依赖关系
    dependency_count = 0
    for task in tasks[:20]:  # 采样检查前20个任务
        deps = self._extract_file_dependencies(task.source_file, task.compile_args)
        dependency_count += len(deps)
    
    avg_deps = dependency_count / min(20, len(tasks))
    
    # 如果平均依赖数 < 0.5，认为是独立任务
    if avg_deps < 0.5:
        self.logger.info(f"检测到独立任务批次 (平均依赖{avg_deps:.1f})，使用性能感知调度")
        return False
    
    self.logger.info(f"检测到依赖任务批次 (平均依赖{avg_deps:.1f})，使用DAG调度")  
    return True

def select_node(self, task: CompileTask, available_nodes: List[ServerNode]) -> SchedulingDecision:
    """智能节点选择"""
    
    # ✅ 对单个任务，直接使用性能感知调度
    if isinstance(task, CompileTask):
        return self._performance_aware_schedule(task, available_nodes)
    
    # 原有DAG逻辑保留给批量调度
    try:
        return self._dag_schedule(task, available_nodes)
    except Exception as e:
        return self._performance_aware_schedule(task, available_nodes)

def _performance_aware_schedule(self, task: CompileTask, available_nodes: List[ServerNode]) -> SchedulingDecision:
    """性能感知调度 - 替代_simple_schedule"""
    
    def calculate_score(node):
        perf_weight = node.get_performance_score()  # 现在会使用静态权重
        load_penalty = node.get_load_ratio()
        
        # ✅ 性能优先，但平衡负载
        score = perf_weight * (1.0 - load_penalty * 0.8)  # 减少负载惩罚权重
        return score
    
    best_node = max(available_nodes, key=calculate_score)
    
    return SchedulingDecision(
        task=task,
        selected_node=best_node,
        algorithm_used="performance_aware",  # ✅ 新算法名
        confidence_score=0.85,
        decision_factors={
            "performance_weight": best_node.get_performance_score(),
            "load_ratio": best_node.get_load_ratio(),
            "calculated_score": calculate_score(best_node),
            "scheduling_strategy": "performance_aware"
        }
    )
```

### 修复3: RoundRobin优化 (3分钟)

```python  
# scheduling_algorithms.py 加权轮转算法
class WeightedRoundRobinScheduler:
    """加权轮转调度器 - 考虑节点性能差异"""
    
    def __init__(self):
        self.name = "weighted_round_robin"
        self.node_weights = {}  # node_id -> weight
        self.current_weights = {}  # node_id -> current_weight
        
    def schedule(self, task: CompileTask, available_nodes: List[ServerNode]) -> SchedulingDecision:
        # 初始化权重
        for node in available_nodes:
            if node.node_id not in self.node_weights:
                self.node_weights[node.node_id] = node.get_performance_score()
                self.current_weights[node.node_id] = 0
        
        # 加权轮转算法
        best_node = None
        max_current_weight = -1
        
        for node in available_nodes:
            if not node.is_available():
                continue
                
            # 增加当前权重
            self.current_weights[node.node_id] += self.node_weights[node.node_id]
            
            # 选择当前权重最高的节点
            if self.current_weights[node.node_id] > max_current_weight:
                max_current_weight = self.current_weights[node.node_id]
                best_node = node
        
        if best_node:
            # 减少被选中节点的权重
            total_weight = sum(self.node_weights.values())
            self.current_weights[best_node.node_id] -= total_weight
        
        return SchedulingDecision(
            task=task,
            selected_node=best_node,
            algorithm_used=self.name,
            confidence_score=0.9,
            decision_factors={
                "node_weight": self.node_weights.get(best_node.node_id, 1.0),
                "current_weight": max_current_weight,
                "algorithm": "weighted_round_robin"
            }
        )
```
